# ============================================================
# Kolla-Ansible Installation with External Ceph Backend
# OpenStack 2024.1 (Caracal) on Ubuntu
# ============================================================

# ========================
# Step 1: Install Dependencies on Deployment Host
# ========================
sudo apt update
sudo apt install -y python3-dev libffi-dev gcc libssl-dev python3-venv git

# Create and activate the virtual environment
python3 -m venv ~/kolla-venv
source ~/kolla-venv/bin/activate

# Upgrade pip and install Kolla-Ansible
pip install -U pip
pip install "ansible>=8,<10"
pip install "kolla-ansible==18.0.0"

# change 2024.1 to 2024.2 if needed in below file
nano kolla-venv/share/kolla-ansible/requirements.yml

kolla-ansible install-deps

# ========================
# Step 2: Prepare Kolla Configuration Directory
# ========================
sudo mkdir -p /etc/kolla
sudo chown $USER:$USER /etc/kolla
cp -r kolla-venv/share/kolla-ansible/etc_examples/kolla/* /etc/kolla/

# Generate passwords
kolla-genpwd

# Copy your custom globals.yml and multinode inventory to /etc/kolla/

# ========================
# Step 3: Setup Ceph Configuration for Kolla
# ========================
# Create service-specific config directories
mkdir -p /etc/kolla/config/glance
mkdir -p /etc/kolla/config/cinder/cinder-volume
mkdir -p /etc/kolla/config/cinder/cinder-backup
mkdir -p /etc/kolla/config/nova

# Copy ceph.conf from your Ceph cluster to all service dirs
CEPH_ADMIN="192.168.10.11"  # One of your Ceph MON nodes
scp $CEPH_ADMIN:/etc/ceph/ceph.conf /etc/kolla/config/glance/
scp $CEPH_ADMIN:/etc/ceph/ceph.conf /etc/kolla/config/cinder/
scp $CEPH_ADMIN:/etc/ceph/ceph.conf /etc/kolla/config/nova/

# Copy keyrings (create these on Ceph cluster first - see ceph-setup.sh)
scp $CEPH_ADMIN:/etc/ceph/ceph.client.glance.keyring /etc/kolla/config/glance/
scp $CEPH_ADMIN:/etc/ceph/ceph.client.cinder.keyring /etc/kolla/config/cinder/cinder-volume/
scp $CEPH_ADMIN:/etc/ceph/ceph.client.cinder-backup.keyring /etc/kolla/config/cinder/cinder-backup/
scp $CEPH_ADMIN:/etc/ceph/ceph.client.nova.keyring /etc/kolla/config/nova/
scp $CEPH_ADMIN:/etc/ceph/ceph.client.cinder.keyring /etc/kolla/config/nova/  # For live migration

# ========================
# Step 4: Deploy OpenStack
# ========================
kolla-ansible -i multinode bootstrap-servers
kolla-ansible -i multinode prechecks
kolla-ansible -i multinode deploy
kolla-ansible -i multinode post-deploy

# ========================
# Troubleshooting Commands
# ========================
# Reconfigure specific service
kolla-ansible -i multinode reconfigure --tags cinder
kolla-ansible -i multinode reconfigure --tags glance
kolla-ansible -i multinode reconfigure --tags nova

# Check service status
docker ps | grep -E "cinder|glance|nova"

# Test Ceph connectivity from container
docker exec -it cinder_volume ceph -s
